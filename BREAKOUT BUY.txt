// ==========================
// 3. Function: Find A Peak And Check B Buy
// A = peak / pivot reference.
// B = current candle buy point.
// ==========================
f_peak_reclaim_signal(_lookback) =>
    float aPrice = na
    int aOffset = na

    for i = 1 to _lookback
        if not na(high[i])
            if na(aPrice) or high[i] > aPrice
                aPrice := high[i]
                aOffset := i

    bool pulledBackBelow20 = false

    for j = 1 to _lookback
        if not na(aOffset) and j < aOffset
            if low[j] < sma20[j]
                pulledBackBelow20 := true

    bool aAbove20Sma = not na(aOffset) and not na(aPrice) and aPrice > sma20[aOffset]
    bool bCrossAboveA = not na(aPrice) and close > aPrice and close[1] <= aPrice
    bool bWithinUpside = not na(aPrice) and close <= aPrice * (1 + maxUpsidePct / 100)

    bool signal = sma200SlopePositive and greenCandle and sma20Above50 and aAbove20Sma and bAbove20Sma and pulledBackBelow20 and bCrossAboveA and bWithinUpside

    int aBarIndex = not na(aOffset) ? bar_index - aOffset : na

    [signal, aPrice, aBarIndex, pulledBackBelow20, bCrossAboveA, bWithinUpside, aAbove20Sma]

// ==========================
// 4. Three Independent Lookback Modules
// ==========================
[n1SignalRaw, n1APrice, n1ABarIndex, n1PulledBack, n1CrossAbove, n1Within, n1AAbove20] = f_peak_reclaim_signal(lookbackN1)
[n2SignalRaw, n2APrice, n2ABarIndex, n2PulledBack, n2CrossAbove, n2Within, n2AAbove20] = f_peak_reclaim_signal(lookbackN2)
[n3SignalRaw, n3APrice, n3ABarIndex, n3PulledBack, n3CrossAbove, n3Within, n3AAbove20] = f_peak_reclaim_signal(lookbackN3)
